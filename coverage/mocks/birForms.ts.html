/**
 * BIR forms and returns (Settings › Accounting & Tax › BIR forms), in BIR's own categories (bir.gov.ph ›
 * BIR Forms). Tax codes link to the return they're filed on, withholding taxes to the forms they're
 * remitted and certified on. Form numbers and titles are BIR's "List of BIR Forms" (bir.gov.ph, 2026),
 * verbatim; the category split is ours, as that list has none. Due dates are as of 2026 (TRAIN, CREATE,
 * EOPT) and come from firm tax calendars and the regulation in `note` ("Deadline source: …"), not BIR text.
 */
export type BirFormCategory =
  | "Application Forms"
  | "Certificates"
  | "Documentary Stamp Tax Return"
  | "Excise Tax Return"
  | "Income Tax Return"
  | "Legal Forms"
  | "Other Forms"
  | "Payment/Remittance Forms"
  | "Transfer Tax Return"
  | "VAT/Percentage Tax Returns";

/** The id of a seeded BIR form, from its number ("2550Q" → "wf-2550q"). Tax codes and withholding taxes link to forms by id. */
export const birFormId = (no: string) => `wf-${no.toLowerCase()}`;
/** The form number ("2550Q"). */
export const birFormNo = (f: Pick<BirForm, 'form'>) => f.form;
 
export interface BirForm {
  id: string;
  /** Form number, e.g. "2550Q". */
  form: string;
  description: string;
  category: BirFormCategory;
  /** Due date when filed through eFPS; '' where eFPS doesn't apply. */
  dueEfps: string;
  /** Due date when filed manually / through eBIRForms. */
  dueManual: string;
  /** Legal basis, or why the form is discontinued. */
  note: string;
  /** Where the due date was checked. */
  source: string;
  active: boolean;
}
 
const form = (category: BirFormCategory, no: string, description: string, dueEfps: string, dueManual: string, note: string, source: string): BirForm => ({
  id: birFormId(no), form: no, description, category, dueEfps, dueManual, note, source, active: true,
});
/** A form BIR no longer uses: kept, inactive, so old references still resolve. */
const discontinued = (...args: Parameters<typeof form>): BirForm => ({ ...form(...args), active: false });
